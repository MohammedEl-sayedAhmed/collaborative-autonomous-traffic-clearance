cmake_minimum_required(VERSION 3.16)
project(caatc_lockstep_plant)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-cmake3 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-sim8 REQUIRED)
add_library(CaatcLockstepPlant SHARED LockstepPlant.cc)
target_link_libraries(CaatcLockstepPlant PRIVATE gz-plugin2::gz-plugin2 gz-plugin2::register gz-sim8::gz-sim8)
install(TARGETS CaatcLockstepPlant LIBRARY DESTINATION lib)
